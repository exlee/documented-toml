### An option an application cannot ship a value for is documented all the
### same: written out as `#:` lines, which are TOML, so they name the key the
### block is about. Such a key is not declared, but it is known, so setting it
### is not reported as an option that does not exist.
###
### Set, it merges like any other key, the `#:` line being the default the
### defaults wrote by hand because there was no live value to take one from.
--- DEF ---
##: The only declared option.
declared = 1

##: How accounts are written:
#: [[accounts]]
#: name = "Personal"
#: host = "imap.example.com"
--- USR ---
declared = 1

[[accounts]]
name = "Mine"
