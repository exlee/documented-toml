### The `=` of keys that follow each other directly are lined up. The tool's
### prose, a recorded default, the person's own comment and blank lines each
### start a new group; a key alone in its group gets one space.
--- DEF ---
protocol = "imap"

##: Server.
host = "imap.example.com"
port = 993
security = "tls"

username = "jane"
--- USR ---
protocol = "imap"
host     = "imap.fastmail.com"
port     = 993
security = "tls"
username = "me"
--- RES ---
protocol = "imap"

##: Server.
#: host = "imap.example.com"
host     = "imap.fastmail.com"
port     = 993
security = "tls"

#: username = "jane"
username = "me"
### Each header opens a span of its own, arrays of tables entry by entry.
--- DEF ---
a = 1
bb = 2
[section]
longer = 3
c = 4
[[items]]
x = 1
yy = 2
--- USR ---
--- RES ---
a  = 1
bb = 2

[section]
longer = 3
c      = 4

[[items]]
x  = 1
yy = 2
### A dotted key is measured by its whole path, indentation included.
--- DEF ---
[s]
a.b = 1
  longer.key.path = 2
c = 3
--- USR ---
--- RES ---
[s]
a.b               = 1
  longer.key.path = 2
c                 = 3
### A value written over several lines keeps its own spacing and sets no width.
--- DEF ---
a = 1
longer = 2
text = '''
x
'''
list = [
  1,
]
--- USR ---
--- RES ---
a      = 1
longer = 2
text = '''
x
'''
list = [
  1,
]
### A person's own padding is recomputed, down to one space for a lone key.
--- DEF ---
a = 1
bb = 2
[one]
only = 1
--- USR ---
a        = 1
bb = 2
[one]
only    = 1
--- RES ---
a  = 1
bb = 2

[one]
only = 1
### `no-align` after `--- DEF ---` keeps the spacing as written.
--- DEF --- no-align
a = 1
longer = 2
--- USR ---
--- RES ---
a = 1
longer = 2
