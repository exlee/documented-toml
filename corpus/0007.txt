### An array in the person's file replaces the default array entirely.
### Elements are not unioned, appended or deduplicated.
--- DEF ---
##: Hosts to try, in order.
hosts = ["a", "b"]
--- USR ---
hosts = ["c"]
--- RES ---
##: Hosts to try, in order.
#: hosts = ["a", "b"]
hosts = ["c"]
### The person's layout of the array survives. Comparison against the default
### strips formatting, so laying the same elements out over several lines is
### not a departure from the default and records nothing.
--- USR ---
hosts = [
  "a",
  "b",
]
--- RES ---
##: Hosts to try, in order.
hosts = [
  "a",
  "b",
]
