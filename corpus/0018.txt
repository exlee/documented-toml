### A block ends where the next anchor begins. A `#:` line opening a table
### header names a key of its own, so what follows documents that key and not
### the last one, and the prose written directly above the header introduces
### it. Each block is read against the section the headers before it opened,
### the way a file of its own would be, so a bare assignment after a header
### lands inside it.
###
### Each key then goes where it belongs: set, above the person's own value;
### unset, left as the `#:` lines it was written as, at the point in the shape
### the defaults gave it.
--- DEF ---
##: The only declared option.
declared = 1

##: How a source is written:
#: [[source]]
#: name = "example"
#:
#: [source.incoming]
#: host = "imap.example.com"
#:
##: Where answers go.
#: [source.outgoing]
#: host = "smtp.example.com"
--- USR ---
declared = 1

[[source]]
name = "mine"

[source.outgoing]
host = "smtp.mine.com"
